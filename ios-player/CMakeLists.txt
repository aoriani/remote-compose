cmake_minimum_required(VERSION 3.20)
project(RemoteComposeIOS LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(SKIA_ARCHIVE "Skia-m144-e2e6623374-4-iosSim-Release-arm64.zip")
set(SKIA_HASH "4065a088f07e2bdf604f9b732550cc6315e8759189eb40ba112051294f18c573")
if(CMAKE_OSX_SYSROOT MATCHES "iphoneos")
  set(SKIA_ARCHIVE "Skia-m144-e2e6623374-4-ios-Release-arm64.zip")
  set(SKIA_HASH "d8f7140066f0637c292b31c1ada8bfec30ff6e229f41e527c9690708b6b8a173")
endif()
FetchContent_Declare(skia URL "https://github.com/JetBrains/skia-pack/releases/download/m144-e2e6623374-4/${SKIA_ARCHIVE}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE URL_HASH "SHA256=${SKIA_HASH}")
FetchContent_GetProperties(skia)
if(NOT skia_POPULATED)
  FetchContent_Populate(skia)
endif()
set(SKIA_DIR "${skia_SOURCE_DIR}")
file(GLOB_RECURSE SKIA_STATIC "${SKIA_DIR}/out/libskia.a" "${SKIA_DIR}/out/*/libskia.a")
list(GET SKIA_STATIC 0 SKIA_STATIC)
get_filename_component(SKIA_LIB_DIR "${SKIA_STATIC}" DIRECTORY)
add_library(skia_lib STATIC IMPORTED GLOBAL)
set_target_properties(skia_lib PROPERTIES IMPORTED_LOCATION "${SKIA_STATIC}" INTERFACE_INCLUDE_DIRECTORIES "${SKIA_DIR}")
set(RC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../upstream/experiments/players/cpp")
add_subdirectory("${RC_ROOT}/lib/rccore" "${CMAKE_BINARY_DIR}/rccore")
add_subdirectory("${RC_ROOT}/lib/rcskia" "${CMAKE_BINARY_DIR}/rcskia")
add_library(rcbridge STATIC Bridge/RCMetalRenderer.mm)
target_include_directories(rcbridge PRIVATE "${SKIA_DIR}")
target_compile_options(rcbridge PRIVATE -fobjc-arc)
target_link_libraries(rcbridge PRIVATE rccore rcskia skia_lib)
